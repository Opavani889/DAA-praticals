n = int(input("Enter a number: "))

fact = 1

for i in range(1, n + 1):
    fact = fact * i

print("Factorial =", fact)

# Time Complexity: O(n)
# Space Complexity: O(1)



def factorial(n):
    if n == 0 or n == 1:
        return 1
    else:
        return n * factorial(n - 1)


n = int(input("Enter a number: "))

result = factorial(n)

print("Factorial =", result)

# Time Complexity: O(n)
