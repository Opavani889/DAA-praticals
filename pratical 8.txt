#include <stdio.h>

#define MAX 10

int graph[MAX][MAX];
int visited[MAX];
int n;

// DFS Traversal
void DFS(int vertex) {
    int i;

    visited[vertex] = 1;
    printf("%d ", vertex);

    for (i = 0; i < n; i++) {
        if (graph[vertex][i] == 1 && visited[i] == 0) {
            DFS(i);
        }
    }
}

// BFS Traversal
void BFS(int start) {
    int queue[MAX];
    int front = 0, rear = 0;
    int visitedBFS[MAX] = {0};
    int current, i;

    queue[rear++] = start;
    visitedBFS[start] = 1;

    while (front < rear) {
        current = queue[front++];
        printf("%d ", current);

        for (i = 0; i < n; i++) {
            if (graph[current][i] == 1 && visitedBFS[i] == 0) {
                queue[rear++] = i;
                visitedBFS[i] = 1;
            }
        }
    }
}

int main() {
    int edges;
    int u, v;
    int start;
    int i, j;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    // Initialize graph
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            graph[i][j] = 0;
        }
    }

    printf("Enter number of edges: ");
    scanf("%d", &edges);

    printf("Enter edges (u v):\n");

    for (i = 0; i < edges; i++) {
        scanf("%d %d", &u, &v);

        graph[u][v] = 1;
        graph[v][u] = 1;  // Undirected graph
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    // DFS
    for (i = 0; i < n; i++) {
        visited[i] = 0;
    }

    printf("\nDFS Traversal: ");
    DFS(start);

    // BFS
    printf("\nBFS Traversal: ");
    BFS(start);

    printf("\n");

    return 0;
}
